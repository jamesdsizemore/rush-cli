def _require_fd_io():
    import os
    if (os.open not in os.supports_dir_fd or os.link not in os.supports_dir_fd
            or not hasattr(os, "O_NOFOLLOW")):
        raise ValueError("race_safe_file_io_unavailable")


def _read_bytes(root, relative, limit=16*1024*1024):
    import os
    from rush.workflows.projects import open_contained_file
    _require_fd_io()
    fd = open_contained_file(root, str(relative))
    try:
        with os.fdopen(fd, "rb", closefd=False) as stream:
            raw = stream.read(limit+1)
        if len(raw)>limit:
            raise ValueError("file_too_large")
        return raw
    finally:
        os.close(fd)


def _dir_fd(root, relative=".", *, create=False):
    import os
    from pathlib import Path
    from rush.io.physical_paths import PhysicalRoot
    _require_fd_io()
    physical=PhysicalRoot(root)
    physical.open_contained(relative, purpose="write")
    flags=os.O_RDONLY|os.O_DIRECTORY|os.O_NOFOLLOW
    fd=os.open(physical.root_path,flags)
    try:
        for part in Path(relative).parts:
            if part==".":
                continue
            if create:
                try:
                    os.mkdir(part,0o700,dir_fd=fd)
                except FileExistsError:
                    pass
            child=os.open(part,flags,dir_fd=fd)
            os.close(fd)
            fd=child
        return fd
    except BaseException:
        os.close(fd)
        raise


def _publish_bytes(root, relative, raw, *, overwrite=False):
    import os,stat,uuid
    from pathlib import Path
    relative=Path(relative);parent=_dir_fd(root,relative.parent)
    token=uuid.uuid4().hex;temp=".rush-publish-"+token;backup=".rush-backup-"+token
    saved=False;published=False;new_identity=None
    def identity(name):
        try:
            item=os.stat(name,dir_fd=parent,follow_symlinks=False)
            return item,(item.st_dev,item.st_ino)
        except FileNotFoundError:
            return None,None
    try:
        current,old_identity=identity(relative.name)
        if current and (not stat.S_ISREG(current.st_mode) or current.st_nlink!=1):
            raise ValueError("output_not_exclusive_regular_file")
        if current and not overwrite:raise FileExistsError(relative.name)
        fd=os.open(temp,os.O_WRONLY|os.O_CREAT|os.O_EXCL|os.O_NOFOLLOW,0o600,dir_fd=parent)
        try:
            new=os.fstat(fd);new_identity=(new.st_dev,new.st_ino)
            with os.fdopen(fd,"wb",closefd=False) as stream:
                stream.write(raw);stream.flush();os.fsync(fd)
        finally:os.close(fd)
        if current:
            os.link(relative.name,backup,src_dir_fd=parent,dst_dir_fd=parent,follow_symlinks=False)
            saved=True
            if identity(backup)[1]!=old_identity or identity(relative.name)[1]!=old_identity:
                raise ValueError("output_changed")
        elif identity(relative.name)[0] is not None:
            raise FileExistsError(relative.name)
        if overwrite:
            os.replace(temp,relative.name,src_dir_fd=parent,dst_dir_fd=parent)
        else:
            os.link(temp,relative.name,src_dir_fd=parent,dst_dir_fd=parent,follow_symlinks=False)
            os.unlink(temp,dir_fd=parent)
        published=True;os.fsync(parent)
        check=_dir_fd(root,relative.parent)
        try:
            if (os.fstat(check).st_dev,os.fstat(check).st_ino)!=(os.fstat(parent).st_dev,os.fstat(parent).st_ino):
                raise ValueError("output_parent_changed")
        finally:os.close(check)
        return Path(root)/relative
    except BaseException:
        # Restore only our new inode; never delete or replace a concurrent writer's leaf.
        if published and identity(relative.name)[1]==new_identity:
            if saved:
                os.replace(backup,relative.name,src_dir_fd=parent,dst_dir_fd=parent);saved=False
            else:os.unlink(relative.name,dir_fd=parent)
        raise
    finally:
        for name in ([backup] if saved else [])+[temp]:
            try:os.unlink(name,dir_fd=parent)
            except FileNotFoundError:pass
        os.close(parent)


def _document(raw):
    import json
    document=json.loads(raw)
    if (not isinstance(document,dict) or document.get("bomFormat")!="CycloneDX"
            or document.get("specVersion") not in {"1.5","1.6"}
            or not isinstance(document.get("components"),list)
            or len(document["components"])>100000
            or any(not isinstance(c,dict) or not isinstance(c.get("name"),str)
                   or not c["name"] for c in document["components"])):
        raise ValueError("unsupported or malformed CycloneDX")
    return document


def read_verified_sbom(path, expected_sha256, *, root):
    import hashlib,re
    from pathlib import Path
    if not isinstance(expected_sha256,str) or not re.fullmatch(r"[0-9a-f]{64}",expected_sha256):
        raise ValueError("expected SHA-256 required")
    path,root=Path(path),Path(root)
    relative=path.relative_to(root) if path.is_absolute() else path
    raw=_read_bytes(root,relative)
    if hashlib.sha256(raw).hexdigest()!=expected_sha256:
        raise ValueError("SBOM digest mismatch")
    return _document(raw)


def generate_inventory(path, *, logical_root, output_root, output_path, overwrite, permissions):
    import hashlib,os,tempfile
    from pathlib import Path
    from rush.engines import ENGINES
    from rush.permissions import ExecutionPermissions,check_permissions
    from rush.tools.common import error_result,skipped_result
    from rush.io.physical_paths import PhysicalRoot
    relative=Path(output_path) if output_path is not None else Path("rush-sbom.json")
    if relative.is_absolute():
        relative=relative.relative_to(output_root)
    target=PhysicalRoot(output_root).open_contained(relative,purpose="write")
    if target.exists() and not overwrite:
        raise FileExistsError(str(target))
    required=ExecutionPermissions(artifact_write=True)
    allowed,missing=check_permissions(required,permissions)
    if not allowed:
        return skipped_result("sbom","cdxgen",", ".join(missing),
            metadata={"execution":{"disposition":"not_run","cause":"permission_denied"}})
    _require_fd_io()
    with tempfile.TemporaryDirectory(prefix="rush-sbom-") as folder:
        fresh=Path(folder)/"bom.json"
        result=run_engine(ENGINES["cdxgen"],path,
            ["--output",str(fresh),"--spec-version","1.6","--no-install-deps","--fail-on-error"],
            tool_name="sbom",permissions=permissions,required_permissions=required)
        result["artifacts"]=[]
        if result["status"]!="ok":
            return result
        try:
            raw=_read_bytes(Path(folder),"bom.json")
            document=_document(raw)
        except (OSError,ValueError,TypeError) as error:
            result.update(status="error",summary="fresh SBOM output invalid",artifacts=[])
            result.setdefault("metadata",{})["reason"]="missing_output" if isinstance(error,FileNotFoundError) else "malformed_output"
            return result
        digest=hashlib.sha256(raw).hexdigest()
        try:
            output=_publish_bytes(output_root,relative,raw,overwrite=overwrite)
            read_verified_sbom(output,digest,root=output_root)
        except (OSError,ValueError,TypeError):
            result.update(status="error",summary="SBOM publication failed",artifacts=[])
            result.setdefault("metadata",{})["reason"]="publication_failed"
            return result
        components=document["components"]
        unknown=sum(not isinstance(c.get("version"),str) or not c["version"] for c in components)
        result["artifacts"]=[str(output)]
        result.setdefault("metadata",{})["sbom"]={
            "path":str(output),"sha256":digest,"component_count":len(components),
            "unversioned_components":unknown,"inventory_empty":not components,"fresh_output":True}
        result["status"]="warn" if unknown or not components else "ok"
        return result

def source_manifest(root):

    from rush.workflows.projects import open_contained_file
    from rush.io.physical_paths import PhysicalRoot

        dirs[:]=sorted(d for d in dirs if d not in {".git",".rush"})
        for name in dirs+sorted(files):
            relative=(Path(folder)/name).relative_to(root)
            checked=physical.open_contained(relative)
            mode=checked.lstat().st_mode
            if stat.S_ISDIR(mode):
                continue
            if not stat.S_ISREG(mode):
                raise ValueError("unsupported_source_type")
            fd=open_contained_file(root,relative.as_posix())

                digest=hashlib.sha256()
                while block:=os.read(fd,1024*1024):

                    digest.update(block)
            finally:
                os.close(fd)

            if len(found)>100000:
                raise ValueError("source_limit_exceeded")
    return dict(sorted(found.items()))

def component_rows(document):
    import json
    from collections import Counter
    rows=[]
    for component in document["components"]:
        if not isinstance(component.get("version"),str) or not component["version"]:
            raise ValueError("incomplete_inventory")
        hashes={}
        for item in component.get("hashes",[]):
            if (not isinstance(item,dict) or not isinstance(item.get("alg"),str)
                    or not isinstance(item.get("content"),str) or not item["content"]):
                raise ValueError("invalid_component_hash")
            if item["alg"] in hashes and hashes[item["alg"]]!=item["content"]:
                raise ValueError("contradictory_component_hash")
            hashes[item["alg"]]=item["content"]

            raise ValueError("invalid_component_identity")

    return Counter(rows)

def catalog_raw_invalid_fields(arguments, schema):
    if not isinstance(arguments, dict) or any(type(k) is not str for k in arguments):
        return ["$"]
    properties = schema.get("properties", {})
    definitions = schema.get("$defs", {})

    def matches(value, shape, depth=0):
        if depth > 16:
            return False
        reference = shape.get("$ref")
        if reference is not None:
            prefix = "#/$defs/"
            return (isinstance(reference, str) and reference.startswith(prefix)
                    and reference[len(prefix):] in definitions
                    and matches(value, definitions[reference[len(prefix):]], depth + 1))
        alternatives = shape.get("anyOf")
        if alternatives is not None:
            return any(matches(value, branch, depth + 1) for branch in alternatives)
        kind = shape.get("type")
        if kind == "null":
            return value is None
        if kind == "boolean":
            return type(value) is bool
        if kind == "integer":
            return type(value) is int
        if kind == "number":
            return type(value) is int or (type(value) is float and math.isfinite(value))
        if kind == "string":
            return type(value) is str
        if kind == "array":
            return type(value) is list and all(
                matches(item, shape.get("items", {}), depth + 1) for item in value)
        if kind == "object":
            return type(value) is dict and all(type(key) is str for key in value)
        if "enum" in shape:
            return any(type(value) is type(item) for item in shape["enum"])
        if not shape:
            return (value is None or type(value) in (str, bool, int)
                    or (type(value) is float and math.isfinite(value))
                    or (type(value) is list and all(matches(v, {}, depth + 1) for v in value))
                    or (type(value) is dict and all(
                        type(k) is str and matches(v, {}, depth + 1)
                        for k, v in value.items())))
        return False

    invalid = set(arguments) - set(properties)
    invalid.update(set(schema.get("required", [])) - set(arguments))
    invalid.update(key for key, value in arguments.items()
                   if key in properties and not matches(value, properties[key]))
    return sorted(invalid)

def exchange_stdio(python, root, env, tool_name, arguments, *, registry_root=None):
    import json, queue, subprocess, threading
    argv=[python, "-m", "rush", "mcp", "serve", "--profile", "full"]
    if registry_root is not None:
        startup=("import sys;from pathlib import Path;from rush.workflows import projects;"
                 "projects.default_data_root=lambda:Path(sys.argv[1]);"
                 "from rush.cli import cli;cli(['mcp','serve','--profile','full'])")
        argv=[python,"-c",startup,str(registry_root)]
    child = subprocess.Popen(
        argv,
        cwd=root, env=env, stdin=subprocess.PIPE, stdout=subprocess.PIPE,
        stderr=subprocess.PIPE, text=True, bufsize=1)
    output = queue.Queue()
    def collect():
        for line in child.stdout:
            output.put(line)
    reader = threading.Thread(target=collect, daemon=True)
    reader.start()
    from collections import deque
    diagnostics=deque(maxlen=1000)
    def drain_stderr():
        for line in child.stderr:
            diagnostics.append(line)
    stderr_reader=threading.Thread(target=drain_stderr,daemon=True)
    stderr_reader.start()
    def send(value):
        child.stdin.write(json.dumps(value) + "\n")
        child.stdin.flush()
    def receive(identity):
        while True:
            item = json.loads(output.get(timeout=1800))
            assert item["jsonrpc"] == "2.0"
            if item.get("id") == identity:
                assert "error" not in item, item
                return item["result"]
    try:
        send({"jsonrpc": "2.0", "id": 1, "method": "initialize",
              "params": {"protocolVersion": "2024-11-05", "capabilities": {},
                         "clientInfo": {"name": "rush-remediation-test", "version": "1"}}})
        initialized = receive(1)
        assert "protocolVersion" in initialized
        send({"jsonrpc": "2.0", "method": "notifications/initialized"})
        send({"jsonrpc": "2.0", "id": 2, "method": "tools/list", "params": {}})
        listed = receive(2)
        schema = next(t["inputSchema"] for t in listed["tools"] if t["name"] == tool_name)
        assert schema["type"] == "object"
        expected={"output_path":(["string","null"],None),
            "overwrite":(["boolean"],False),
            "compare_inventories":(["boolean"],False),
            "runtime_path":(["string","null"],None),
            "image_ref":(["string","null"],None)}
        grants={"allow_network","allow_download","allow_cache_write","allow_build","allow_slow","allow_artifact_write","allow_browser"}
        injected={"project","result_view","limit","max_bytes","no_cache"}
        assert set(schema["properties"])=={"path"}|set(expected)|grants|injected
        assert schema["required"]==["path"]
        assert schema["additionalProperties"] is False
        for name,(types,default) in expected.items():
            node=schema["properties"][name]
            assert {part.get("type") for part in node.get("anyOf",[node])}==set(types)
            assert node["default"]==default
        send({"jsonrpc": "2.0", "id": 3, "method": "tools/call",

        envelope = receive(3)

        content = envelope.get("structuredContent")
        if content is None:

        return content
    finally:
        child.terminate()
        child.wait(timeout=10)
        child.stdin.close()
        child.stdout.close()
        child.stderr.close()
